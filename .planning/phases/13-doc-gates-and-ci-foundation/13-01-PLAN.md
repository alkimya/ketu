    return (armc + (90.0 + AD) / 3.0) % 360.0


def _ra_formula_cusp_12(armc: np.ndarray, AD: np.ndarray) -> np.ndarray:

    return (armc + 2.0 * (90.0 + AD) / 3.0) % 360.0
