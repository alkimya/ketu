import json
from pathlib import Path

_FIXTURES_DIR = Path(__file__).parent / "fixtures"


def load_oracle_fixture(slug: str) -> dict:
    """Load a synastry oracle fixture by slug.

    Parameters
    ----------
    slug : str
        Fixture slug (e.g., 'curie', 'diana_charles', 'lennon_ono').

    Returns
    -------
    dict
        Parsed JSON fixture with keys: name, rodden_a, rodden_b, chart_a,
        chart_b, expected_aspects, validation_source, tolerance_deg.
    """
    path = _FIXTURES_DIR / f"oracle_{slug}.json"
    with path.open() as fh:
        return json.load(fh)


ORACLE_SLUGS = ["curie", "diana_charles", "lennon_ono"]


@pytest.fixture(params=ORACLE_SLUGS, ids=ORACLE_SLUGS)
def oracle_fixture(request):
    """Yield each oracle fixture in turn, parametrized over the 3 couples."""
    return load_oracle_fixture(request.param)
