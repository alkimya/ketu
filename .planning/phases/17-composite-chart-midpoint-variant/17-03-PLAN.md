```python
"""COMP-04 / ROADMAP success criterion #3 — composite oracle tests.

Three reference composite pairs are pinned as regression fixtures with
documented per-body tolerances. Self-consistency is the primary
validation (machine-precision round-trip via compute_chart +
calculate_composite). Cross-validation against Astro.com is deferred
to Plan 17-04 close-out — Astro.com is bot-blocked from automated
retrieval AND defaults to the reference-place method whose ASC/MC
differ from the pure midpoint method (17-RESEARCH §Pitfall 5).

The headline output of pytest -v -s is the max |delta| per fixture —
required by ROADMAP criterion #3 ("documented max longitude delta").
"""
from __future__ import annotations

import json
from pathlib import Path

import numpy as np
import pytest

from ketu.charts import compute_chart
from ketu.composite import calculate_composite
from ketu.ephemeris.time import iso_to_jd  # or whichever helper the project uses

from .conftest import ORACLE_SLUGS, load_oracle_fixture

BODY_NAMES = [
    "Sun", "Moon", "Mercury", "Venus", "Mars", "Jupiter", "Saturn",
    "Uranus", "Neptune", "Pluto", "Rahu", "Ketu", "Lilith",
]  # D-08 frozen axis order
BODY_INDEX = {name: i for i, name in enumerate(BODY_NAMES)}


def _build_charts(fixture: dict):
    """Construct (chart_a, chart_b) by replaying compute_chart on fixture data."""
    a = fixture["chart_a"]
    b = fixture["chart_b"]
    jd_a = iso_to_jd(a["iso_date"])
    jd_b = iso_to_jd(b["iso_date"])
    chart_a = compute_chart(jd_a, a["lat"], a["lon"])
    chart_b = compute_chart(jd_b, b["lat"], b["lon"])
    return chart_a, chart_b


class TestComposeOracleBodies:
    """Per-body composite longitudes match the fixture within tolerance_deg."""

    def test_body_lons_match_oracle(self, oracle_fixture, capsys):
        chart_a, chart_b = _build_charts(oracle_fixture)
        composite = calculate_composite(chart_a, chart_b)

        deltas = []
        for body_name, spec in oracle_fixture["expected_composite"]["body_lons"].items():
            idx = BODY_INDEX[body_name]
            expected = spec["deg"]
            tolerance = spec["tolerance_deg"]
            actual = float(composite["body_lons"][idx])

            # Circular delta on [0, 360)
            raw = abs(actual - expected)
            delta = min(raw, 360.0 - raw)
            deltas.append((body_name, delta, tolerance))

            assert delta <= tolerance, (
                f"{oracle_fixture['name']} {body_name}: expected {expected}°, "
                f"got {actual}°, delta {delta:.6f}° > tolerance {tolerance}°"
            )

        max_body, max_delta, max_tol = max(deltas, key=lambda x: x[1])
        with capsys.disabled():
            print(
                f"\n[oracle:{oracle_fixture['name']}] max body delta: "
                f"{max_delta:.6f}° on {max_body} (tolerance {max_tol}°)"
            )


class TestComposeOracleAngles:
    """ASC and MC match the fixture for AA-rated pairs only."""

    def test_asc_matches_oracle_if_present(self, oracle_fixture):
        if "asc" not in oracle_fixture["expected_composite"]:
            pytest.skip(
                f"{oracle_fixture['name']}: asc excluded (Rodden rating "
                f"uncertainty — see fixture notes)"
            )
        chart_a, chart_b = _build_charts(oracle_fixture)
        composite = calculate_composite(chart_a, chart_b)
        expected = oracle_fixture["expected_composite"]["asc"]["deg"]
        tolerance = oracle_fixture["expected_composite"]["asc"]["tolerance_deg"]
        actual = float(composite["asc"])
        raw = abs(actual - expected)
        delta = min(raw, 360.0 - raw)
        assert delta <= tolerance, (
            f"{oracle_fixture['name']} asc: expected {expected}°, got {actual}°, "
            f"delta {delta:.6f}° > tolerance {tolerance}°"
        )

    def test_mc_matches_oracle_if_present(self, oracle_fixture):
        if "mc" not in oracle_fixture["expected_composite"]:
            pytest.skip(
                f"{oracle_fixture['name']}: mc excluded (Rodden rating "
                f"uncertainty — see fixture notes)"
            )
        chart_a, chart_b = _build_charts(oracle_fixture)
        composite = calculate_composite(chart_a, chart_b)
        expected = oracle_fixture["expected_composite"]["mc"]["deg"]
        tolerance = oracle_fixture["expected_composite"]["mc"]["tolerance_deg"]
        actual = float(composite["mc"])
        raw = abs(actual - expected)
        delta = min(raw, 360.0 - raw)
        assert delta <= tolerance, (
            f"{oracle_fixture['name']} mc: expected {expected}°, got {actual}°, "
            f"delta {delta:.6f}° > tolerance {tolerance}°"
        )


class TestOracleSchemaIntegrity:
    """Fixture schema is stable; cross-check field exists and is recorded as deferred."""

    def test_schema_version_is_1(self, oracle_fixture):
        assert oracle_fixture["schema_version"] == 1

    def test_validation_source_documents_self_consistency(self, oracle_fixture):
        vs = oracle_fixture["validation_source"]
        assert "self-consistency" in vs.lower(), vs

    def test_cross_check_astro_com_deferred(self, oracle_fixture):
        cc = oracle_fixture["cross_check_astro_com"]
        # Plan 17-03 ships with performed=false; Plan 17-04 may flip this to true.
        assert "performed" in cc
        assert "tolerance_deg" in cc
        # tolerance for the cross-check is 0.1° (advisory; not the headline gate).
        assert cc["tolerance_deg"] == 0.1
```
