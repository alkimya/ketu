**Step 5: Verify wheel contents**
```bash
# Check py.typed is included
unzip -l dist/ketu-1.0.0-py3-none-any.whl | grep py.typed
# Must show ketu/py.typed

# Check no removed modules are present
unzip -l dist/ketu-1.0.0-py3-none-any.whl | grep -E "(chart|icalendar|export)" || echo "GOOD: No removed modules found"
# Must print "GOOD: No removed modules found"

# Check all expected packages are present
unzip -l dist/ketu-1.0.0-py3-none-any.whl | grep -c "__init__.py"
# Must show 6 (ketu, ephemeris, aspects, cycles, cache, + METADATA dirs)
```

**Step 6: Test installation in fresh venv**
```bash
# Create temp venv
python -m venv /tmp/test-ketu-install
source /tmp/test-ketu-install/bin/activate

# Install from wheel
pip install dist/ketu-1.0.0-py3-none-any.whl

# Verify version
python -c "import ketu; print(f'Version: {ketu.__version__}'); assert ketu.__version__ == '1.0.0', 'Version mismatch!'"

# Verify core imports work
python -c "from ketu import bodies, aspects, signs; print(f'Bodies: {len(bodies)}, Aspects: {len(aspects)}, Signs: {len(signs)}')"

# Verify submodule imports work
python -c "from ketu.calculations import long, lat, dist_au; print('calculations: OK')"
python -c "from ketu.cycles import generate_cycle_series; print('cycles: OK')"
python -c "from ketu.complex import to_complex, to_polar; print('complex: OK')"

# Verify removed modules are NOT importable
python -c "
try:
    from ketu.export import chart
    print('FAIL: chart module should not exist')
    exit(1)
except (ImportError, ModuleNotFoundError):
    print('GOOD: chart module correctly removed')
"

# Clean up
deactivate
rm -rf /tmp/test-ketu-install
```

**Step 7: Run full test suite one final time**
```bash
cd /home/loc/workspace/solaris/ketu
source venv/bin/activate
```
